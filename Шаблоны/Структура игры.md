# Общая архитектура

Персонаж в Godot лучше реализовать как **иерархию сцен и скриптов**, чтобы можно было быстро настраивать и переиспользовать логику.

---

## 🎬 Иерархия сцен

`Player.tscn 
├── CharacterBody2D        # Главный узел персонажа 
│   ├── CollisionShape2D   # Коллайдер 
│   ├── AnimatedSprite2D   # Спрайт или SpriteFrames 
│   ├── Timer (dash_timer) # Для рывка 
│   ├── Timer (attack_timer) 
│   ├── Marker2D (projectile_spawn) 
│   ├── Area2D (attack_area) 
│   └── Particles2D / Trail  # Эффект рывка 
└── Scripts/     
	├── Player.gd     
	├── PlayerStats.gd     
	├── PlayerMovement.gd     
	├── PlayerCombat.gd     
	└── PlayerEffects.gd`

---

# 📄 Классы и их назначение

---

## 🧍‍♂️ **Player.gd**

Главный контроллер, объединяющий все подсистемы.

> `extends CharacterBody2D 
> class_name Player  
> @onready 
> var stats = PlayerStats.new() @onready var movement = PlayerMovement.new(self) 
> @onready var combat = PlayerCombat.new(self) @onready var effects = PlayerEffects.new(self)  
> var state : String = "idle"  
> func _ready():     
> 	movement.connect("dash_started", self._on_dash_started)     combat.connect("attack_started", self._on_attack_started)  
> func _physics_process(delta):     
> 	movement.update(delta)     
> 	combat.update(delta)     
> 	effects.update(delta)  
> 
> func _on_dash_started():     # Можно добавить анимацию, звук, эффект     $Trail.emitting = true  
> 
> func _on_attack_started():     # Можно вызвать эффект удара или световой всплеск     
> 	effects.play_flash()

## 🧮 **PlayerStats.gd**

Хранит параметры персонажа — "модель данных".

`extends Resource class_name PlayerStats  var max_health : int = 5 var health : int = 5 var energy : float = 3.0 var max_energy : float = 3.0  var move_speed : float = 120.0 var dash_speed : float = 400.0 var dash_duration : float = 0.15 var dash_cooldown : float = 0.5  var attack_damage : int = 1 var attack_range : float = 40.0 var attack_cooldown : float = 0.3`

> 💡 _PlayerStats можно сделать `Resource`, чтобы легко настраивать разные персонажи/вариации через инспектор._

---

## 🏃 **PlayerMovement.gd**

Отвечает за передвижение, рывки и направление взгляда.

`extends Node class_name PlayerMovement  signal dash_started signal dash_finished  var player : CharacterBody2D var velocity : Vector2 = Vector2.ZERO var dash_time : float = 0.0 var dash_on_cooldown : bool = false var dash_dir : Vector2 = Vector2.ZERO  func _init(p):     player = p  func update(delta):     if Input.is_action_just_pressed("dash") and not dash_on_cooldown:         start_dash()     elif dash_time > 0:         apply_dash(delta)     else:         apply_move(delta)  func apply_move(delta):     var dir = Input.get_vector("move_left", "move_right", "move_up", "move_down")     velocity = dir * player.stats.move_speed     player.velocity = velocity     player.move_and_slide()     if dir != Vector2.ZERO:         player.look_at(player.global_position + dir)  func start_dash():     dash_dir = Input.get_vector("move_left", "move_right", "move_up", "move_down").normalized()     if dash_dir == Vector2.ZERO:         dash_dir = player.get_global_transform().x.normalized() # текущее направление     dash_time = player.stats.dash_duration     dash_on_cooldown = true     emit_signal("dash_started")  func apply_dash(delta):     dash_time -= delta     player.velocity = dash_dir * player.stats.dash_speed     player.move_and_slide()     if dash_time <= 0:         emit_signal("dash_finished")         await get_tree().create_timer(player.stats.dash_cooldown).timeout         dash_on_cooldown = false`

## ⚔️ **PlayerCombat.gd**

Атака, выстрелы и взаимодействие с врагами.

`extends Node class_name PlayerCombat  signal attack_started  var player : CharacterBody2D var attack_cooldown : float = 0.0  func _init(p):     player = p  func update(delta):     if attack_cooldown > 0:         attack_cooldown -= delta     if Input.is_action_just_pressed("attack") and attack_cooldown <= 0:         perform_attack()  func perform_attack():     emit_signal("attack_started")     attack_cooldown = player.stats.attack_cooldown     var enemies = player.$attack_area.get_overlapping_bodies()     for e in enemies:         if e.has_method("take_damage"):             e.take_damage(player.stats.attack_damage)`

## ✨ **PlayerEffects.gd**

Визуальные и звуковые эффекты — вспышки, следы, экраны.

`extends Node class_name PlayerEffects  var player : CharacterBody2D  func _init(p):     player = p  func update(delta):     # Можно реализовать эффекты post-process при низком HP     pass  func play_flash():     # Визуальный флэш при атаке     if player.has_node("AnimatedSprite2D"):         var sprite = player.get_node("AnimatedSprite2D")         sprite.modulate = Color(1, 0.8, 0.8)         await get_tree().create_timer(0.1).timeout         sprite.modulate = Color(1, 1, 1)`

# 💡 Принципы архитектуры

- **Разделение ответственности** — движение, бой и эффекты независимы.
    
- **Композиция, не наследование** — Player агрегирует логику.
    
- **Сигналы** связывают модули (например, `dash_started` → эффекты).
    
- **Resource Stats** позволяет создавать врагов/вариации игрока через переиспользование.
