## День 2: Сетка и юниты

### Шаг 1. GridUtils.gd

```gdscript
# GridUtils.gd
extends Node

const GRID_WIDTH: int = 10
const GRID_HEIGHT: int = 10
const CELL_SIZE: float = 1.0

static func world_to_grid(world_pos: Vector3) -> Vector2i:
    return Vector2i(int(world_pos.x / CELL_SIZE), int(world_pos.z / CELL_SIZE))

static func grid_to_world(grid_pos: Vector2i) -> Vector3:
    return Vector3(grid_pos.x * CELL_SIZE + CELL_SIZE * 0.5, 0.5, grid_pos.y * CELL_SIZE + CELL_SIZE * 0.5)

static func is_valid_cell(grid_pos: Vector2i) -> bool:
    return grid_pos.x >= 0 and grid_pos.x < GRID_WIDTH and grid_pos.y >= 0 and grid_pos.y < GRID_HEIGHT

static func manhattan_distance(from: Vector2i, to: Vector2i) -> int:
    return absi(from.x - to.x) + absi(from.y - to.y)

static func chebyshev_distance(from: Vector2i, to: Vector2i) -> int:
    return maxi(absi(from.x - to.x), absi(from.y - to.y))
```

### Шаг 2. Unit.gd (базовый класс)

```gdscript
# Unit.gd
extends Node3D
class_name Unit

@export var unit_name: String = "Unit"
@export var max_hp: int = 100
@export var max_stamina: int = 8
@export var stamina_regen: int = 4

var hp: int
var stamina: int
var grid_position: Vector2i
var is_player_controlled: bool = false

func _ready() -> void:
    hp = max_hp
    stamina = max_stamina

func set_grid_position(pos: Vector2i) -> void:
    if GridUtils.is_valid_cell(pos):
        grid_position = pos
        position = GridUtils.grid_to_world(pos)

func take_damage(amount: int) -> void:
    hp -= amount
    hp = max(hp, 0)
    print("%s takes %d damage. HP: %d/%d" % [unit_name, amount, hp, max_hp])

    if hp == 0:
        die()

func die() -> void:
    print("%s is defeated!" % unit_name)
    visible = false

func restore_stamina() -> void:
    stamina = min(stamina + stamina_regen, max_stamina)
```

## День 3: Таймлайн и управление

### Шаг 1. TurnManager.gd

```gdscript
# TurnManager.gd
extends Node
class_name TurnManager

signal turn_started(unit: Unit)
signal turn_ended()

var units: Array[Unit] = []
var turn_order: Array[Unit] = []
var current_turn_index: int = 0
var current_unit: Unit

func _ready() -> void:
    # Собрать всех юнитов из сцены
    units = get_tree().get_nodes_in_group("units")
    calculate_initiative()
    start_next_turn()

func calculate_initiative() -> void:
    var init_data = []
    for unit in units:
        var roll = randi_range(1, 6)
        var total = roll + 6  # Базовая инициатива = 6
        init_data.append({"unit": unit, "init": total})

    init_data.sort_custom(func(a, b): return a["init"] > b["init"])
    turn_order = init_data.map(func(d): return d["unit"])

    print("Turn order:")
    for i in range(turn_order.size()):
        print("  %d. %s" % [i + 1, turn_order[i].unit_name])

func start_next_turn() -> void:
    current_unit = turn_order[current_turn_index]
    current_unit.restore_stamina()
    turn_started.emit(current_unit)

    print("\n=== %s's turn ===" % current_unit.unit_name)

    if current_unit.is_player_controlled:
        print("Waiting for player input...")
    else:
        # AI ходит автоматически (позже)
        await get_tree().create_timer(1.0).timeout
        ai_take_turn()

func ai_take_turn() -> void:
    # Простой AI: случайная атака
    print("AI choosing action...")
    await get_tree().create_timer(0.5).timeout
    end_turn()

func end_turn() -> void:
    turn_ended.emit()
    current_turn_index = (current_turn_index + 1) % turn_order.size()
    await get_tree().create_timer(0.3).timeout
    start_next_turn()
```

### Шаг 2. PlayerInput.gd

```gdscript
# PlayerInput.gd
extends Node

@onready var turn_manager = get_node("/root/BattleScene/TurnManager")

var selected_cell: Vector2i = Vector2i(-1, -1)
var attacking: bool = false

func _input(event: InputEvent) -> void:
    if not turn_manager.current_unit.is_player_controlled:
        return

    if event is InputEventKey and event.pressed:
        match event.keycode:
            KEY_W: move_unit(Vector2i(0, -1))
            KEY_S: move_unit(Vector2i(0, 1))
            KEY_A: move_unit(Vector2i(-1, 0))
            KEY_D: move_unit(Vector2i(1, 0))
            KEY_SPACE: end_player_turn()

func move_unit(direction: Vector2i) -> void:
    var current_unit = turn_manager.current_unit
    var new_pos = current_unit.grid_position + direction

    if GridUtils.is_valid_cell(new_pos):
        current_unit.set_grid_position(new_pos)
        current_unit.stamina -= 1
        print("Moved to %s. Stamina: %d/%d" % [new_pos, current_unit.stamina, current_unit.max_stamina])

func end_player_turn() -> void:
    print("Player ends turn")
    turn_manager.end_turn()
```

## День 4: Карточная система и урон

### Шаг 1. Ability.gd

```gdscript
# Ability.gd
extends Resource
class_name Ability

@export var ability_name: String = "Attack"
@export var stamina_cost: int = 2
@export var damage_base: int = 10
@export var damage_type: String = "PHYSICAL"
@export var range_tiles: int = 1
@export var is_aoe: bool = false
@export var aoe_radius: int = 0
@export var description: String = ""

func apply(attacker: Unit, defender: Unit) -> int:
    var damage = DamageCalculator.calculate_damage(attacker, self, defender)
    defender.take_damage(damage)
    return damage
```

### Шаг 2. DamageCalculator.gd

```gdscript
# DamageCalculator.gd
extends Node
class_name DamageCalculator

static func calculate_damage(attacker: Unit, ability: Ability, defender: Unit) -> int:
    var base_damage = ability.damage_base
    var critical_roll = randf()
    var is_critical = critical_roll < 0.05  # 5% крит шанс

    var final_damage = base_damage
    if is_critical:
        final_damage = int(final_damage * 1.5)
        print("CRITICAL HIT!")

    # Броня
    final_damage -= defender.max_hp / 20  # Упрощённо: броня = max_hp / 20

    return maxi(final_damage, 1)
```

### Шаг 3. CardHand.gd

```gdscript
# CardHand.gd
extends Node
class_name CardHand

var ability_pool: Array[Ability] = []
var current_hand: Array[Ability] = []

func _ready() -> void:
    # Создать простые способности
    var attack = Ability.new()
    attack.ability_name = "Attack"
    attack.stamina_cost = 1
    attack.damage_base = 12

    var power_attack = Ability.new()
    power_attack.ability_name = "Power Attack"
    power_attack.stamina_cost = 3
    power_attack.damage_base = 22

    ability_pool = [attack, power_attack, attack, attack, attack]
    draw_hand()

func draw_hand() -> void:
    current_hand.clear()
    for i in range(5):
        var ability = ability_pool[randi() % ability_pool.size()]
        current_hand.append(ability)

func use_ability(index: int, attacker: Unit, defender: Unit) -> bool:
    if index < 0 or index >= current_hand.size():
        return false

    var ability = current_hand[index]

    if attacker.stamina < ability.stamina_cost:
        print("Not enough stamina!")
        return false

    print("%s uses %s!" % [attacker.unit_name, ability.ability_name])
    attacker.stamina -= ability.stamina_cost
    ability.apply(attacker, defender)

    draw_hand()  # Новая рука
    return true
```

## День 5: UI (HP бары, логи, карты)

### Шаг 1. HUD.gd

```gdscript
# HUD.gd
extends CanvasLayer

@onready var log_label = Label.new()
@onready var card_container = HBoxContainer.new()
@onready var stamina_label = Label.new()

var log_messages: Array[String] = []

func _ready() -> void:
    # Логирование боя
    log_label.text = "Battle started!\n"
    add_child(log_label)

    # Карты
    add_child(card_container)
    card_container.anchor_left = 0.0
    card_container.anchor_bottom = 1.0
    card_container.anchor_right = 1.0
    card_container.offset_bottom = -40

func log_message(message: String) -> void:
    log_messages.append(message)
    if log_messages.size() > 10:
        log_messages.pop_front()

    log_label.text = "\n".join(log_messages)

func update_card_display(hand: Array[Ability]) -> void:
    card_container.queue_free_children()

    for i in range(hand.size()):
        var card_label = Label.new()
        card_label.text = "[%d] %s" % [i + 1, hand[i].ability_name]
        card_container.add_child(card_label)
```
