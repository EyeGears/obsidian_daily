extends CharacterBody3D


 
@onready var head = $head
@onready var cam = $head/Camera3D



var accel = 6
var SPEED = 5.0
var JUMP_VELOCITY = 2.5
var crouched = false # определяет приседает игрок или нет
var input_dir = Vector3(0,0,0) # направление нажатия кнопок
var direction = Vector3() # направление игрок
var sens = 0.005
var gravity = ProjectSettings.get_setting("physics/3d/default_gravity")

var x = 0
var x1 = 2
var pr = 0
var bust = 300

func _ready():
	set_process_input(true)
	Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)
 
func _input(event: InputEvent): #повороты мышкой
	if Input.is_action_just_pressed("ui_cancel"):
		Input.set_mouse_mode(Input.MOUSE_MODE_VISIBLE)
	
		#$head/Camera3D/AnimationPlayer.play("run")
	if Input.get_mouse_mode() == Input.MOUSE_MODE_CAPTURED:
		if event is InputEventMouseMotion:
			head.rotate_y(-event.relative.x * sens)
			cam.rotate_x(-event.relative.y * sens)
			cam.rotation.x = clamp(cam.rotation.x,deg_to_rad(-89),deg_to_rad(89))
			
				
			
			
func _physics_process(delta):
	print(SPEED)
	#print(cam.rotation_degrees.x)
	#is_colliding
	if pr > 50 and is_on_floor():
		cam.rotation_degrees.x = cam.rotation_degrees.x - 6.5
		set_process_input(false)
		if cam.rotation_degrees.x <= -360:
			cam.rotation_degrees.x = 0
			pr = 0
			set_process_input(true)
	
	
	if $head/Camera3D/RayCast3D.is_colliding():
		var object = $head/Camera3D/RayCast3D.get_collider()
		if object.name == "BLOCK1":
			x = x + 1
			print(x)
		if object.name == "BLOCK2":
			x = x - 1
			print(x)
		if object.name == "BOX":
			x = x + 1
			print(x)
		if object.name == "BOX2":
			x = x - 1
			print(x)
		#x = x + 1
		#print(x)
		

		
	if Input.is_action_pressed("+crouch"): # приседание
		crouched = true
		SPEED = 2.5
		$BodyPlayer.scale.y = lerp($BodyPlayer.scale.y,0.4,0.4)
		$BodyPlayer.position.y = lerp($BodyPlayer.position.y, 0.66,0.4)
		head.position.y = lerp(head.position.y, 0.5, 0.3)
		
	else:
		crouched = false
		SPEED = 5
		$BodyPlayer.scale.y = lerp($BodyPlayer.scale.y, 1.0 ,0.4)
		$BodyPlayer.position.y = lerp($BodyPlayer.position.y, 1.143,0.4)
		head.position.y = lerp(head.position.y, 1.85 , 0.3)
 
	if not is_on_floor(): # гравитация
		velocity.y -= gravity * delta
		pr = pr + 1
		
 
	if Input.is_action_just_pressed("ui_accept") and is_on_floor() and crouched == false: # прыжок
		velocity.y = JUMP_VELOCITY
		
	if is_on_floor() and Input.is_action_pressed("+shift"):
		SPEED = 10
		
		if Input.is_action_pressed("+crouch") and SPEED == 10:
			SPEED = bust
			x1 = x1 + 2.5
			SPEED = SPEED / x1
		if Input.is_action_just_released("+crouch"):
			x1 = 2
	else:
		SPEED = 5
		x1 = 2
	
	
	
		#$head/Camera3D/AnimationPlayer.play("run")
		
		
		#if $head/Camera3D.rotation.x == -360:
			#$head/Camera3D.rotation.x = 0
			#pr = 0
	
	if pr <= 50 and is_on_floor():
		pr = 0
	################################################хождение туда сюда
	input_dir = Input.get_vector("+a", "+d", "+w", "+s")
	direction = ($head.transform.basis * Vector3(input_dir.x, 0, input_dir.y)).normalized()

	velocity.x = lerp(velocity.x ,direction.x * SPEED, accel * delta)
	velocity.z = lerp(velocity.z ,direction.z * SPEED, accel * delta)
	move_and_slide()
